<script>
const FILES=JSON.parse('{{filelist}}');
const CSRF='{{csrf}}';
const currentDir='{{currentDir}}';
const BASE_URL=location.pathname+'?of=usrz7'+(currentDir?('&dir='+currentDir):'');
const listEl=document.getElementById('filelist');
document.getElementById('filebtn').onclick=_=>document.getElementById('fileinput').click();
document.getElementById('fileinput').onchange=_=>{
  if(!document.getElementById('fileinput').files.length)return;
  document.getElementById('filebtn').textContent=document.getElementById('fileinput').files[0].name;
}
function renderList(){
  listEl.innerHTML=FILES.map(f=>`<div class="item">
  <span>${f}</span>
  <span class="actions">
    <a href="download.php?file=${encodeURIComponent(f)}&dir=${currentDir}">Download</a>
    <form method="post" style="display:inline"><input type="hidden" name="file" value="${f}">
    <input type="hidden" name="csrf" value="${CSRF}"><input type="hidden" name="action" value="delete">
    <button type="submit">Delete</button></form>
  </span>
  </div>`).join('');
}
function uploadFile(){
  const f=document.getElementById('fileinput').files[0];if(!f)return;
  const fd=new FormData();fd.append('file',f);fd.append('csrf',CSRF);fd.append('action','upload');
  fetch(BASE_URL,{method:'POST',body:fd}).then(r=>r.text()).then(_=>location.reload());
}
renderList();
</script>
