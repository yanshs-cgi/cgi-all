<script>
const FILES=JSON.parse('{{filelist}}');
const currentDir='{{currentDir}}';
const listEl=document.getElementById('filelist');
const previewEl=document.getElementById('preview');

function renderList(){
  listEl.innerHTML=FILES.map(f=>`<div class="item">
    <span>${f}</span>
    <div class="actions">
      <a href="download.php?file=${encodeURIComponent(f)}&dir=${currentDir}" target="_blank">Download</a>
      <form method="post" style="display:inline" action="filemanager.php?of=usrz7&dir=${currentDir}">
        <input type="hidden" name="file" value="${f}">
        <input type="hidden" name="csrf" value="{{csrf}}">
        <input type="hidden" name="action" value="delete">
        <button type="submit">Delete</button>
      </form>
    </div>
  </div>`).join('');

  document.querySelectorAll('#filelist .item span').forEach(el=>{
    el.addEventListener('click',()=>{
      const name=el.textContent;
      const ext=name.split('.').pop().toLowerCase();
      if(['png','jpg','jpeg','webp'].includes(ext)){
        previewEl.innerHTML=`<img src="download.php?file=${encodeURIComponent(name)}&dir=${currentDir}" style="max-width:100%;max-height:100%;border-radius:8px"/>`;
      } else if(ext==='pdf'){
        previewEl.innerHTML=`<iframe src="download.php?file=${encodeURIComponent(name)}&dir=${currentDir}" style="width:100%;height:100%;border:0;border-radius:8px"></iframe>`;
      } else {
        previewEl.textContent='Tidak ada preview untuk file ini';
      }
    });
  });
}

renderList();
</script>
